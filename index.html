Privacy Architecture

ProofPass never stores sensitive personal information directly onchain. Instead, it utilizes cryptographic commitments: Commitment = Hash(Secret || Data).

For age verification, we implement a genuine cryptographic hash chain. The issuer hashes a random seed N times (Max Age). To prove age >= 18, the holder reveals the (Age - 18)th hash. The verifier hashes this 18 times and compares it to the public commitment.

BOT Chain Smart Contracts

The core Solidity architecture powering ProofPass. These contracts handle issuer whitelisting, credential commitments, and the revocation registry.

ProofPassRegistry.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

/**
 * @title ProofPassRegistry
 * @dev Manages Issuers, Credential Commitments, and Revocations on BOT Chain.
 * Privacy preservation is achieved by ONLY storing cryptographic hashes (commitments).
 */
contract ProofPassRegistry {
    address public admin;

    struct IssuerInfo {
        bool isAuthorized;
        string didDocument; // Decentralized Identifier document URI
        uint256 registeredAt;
    }

    // Mappings
    mapping(address => IssuerInfo) public issuers;
    mapping(bytes32 => address) public credentialToIssuer;
    mapping(bytes32 => bool) public isRevoked;

    // Events
    event IssuerAuthorized(address indexed issuer, string did);
    event CredentialCommitted(bytes32 indexed commitment, address indexed issuer);
    event CredentialRevoked(bytes32 indexed commitment);

    modifier onlyAdmin() {
        require(msg.sender == admin, "ProofPass: Not admin");
        _;
    }

    modifier onlyIssuer() {
        require(issuers[msg.sender].isAuthorized, "ProofPass: Not authorized issuer");
        _;
    }

    constructor() {
        admin = msg.sender;
    }

    /**
     * @dev Authorize a new credential issuer (e.g., a University or Gov body)
     */
    function authorizeIssuer(address _issuer, string memory _did) external onlyAdmin {
        issuers[_issuer] = IssuerInfo(true, _did, block.timestamp);
        emit IssuerAuthorized(_issuer, _did);
    }

    /**
     * @dev Commit a credential to the chain. 
     * @param commitment A cryptographic hash: hash(holderAddress + salt + claimData)
     */
    function commitCredential(bytes32 commitment) external onlyIssuer {
        require(credentialToIssuer[commitment] == address(0), "ProofPass: Already exists");
        credentialToIssuer[commitment] = msg.sender;
        emit CredentialCommitted(commitment, msg.sender);
    }

    /**
     * @dev Revoke a previously issued credential
     */
    function revokeCredential(bytes32 commitment) external onlyIssuer {
        require(credentialToIssuer[commitment] == msg.sender, "ProofPass: Not your credential");
        require(!isRevoked[commitment], "ProofPass: Already revoked");
        
        isRevoked[commitment] = true;
        emit CredentialRevoked(commitment);
    }

    /**
     * @dev Verifier function to check valid state
     */
    function isValid(bytes32 commitment) external view returns (bool) {
        return (credentialToIssuer[commitment] != address(0)) && (!isRevoked[commitment]);
    }
}
